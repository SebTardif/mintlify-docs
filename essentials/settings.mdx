Every Mintlify site needs a `docs.json` file with the core configuration settings. Learn more about the [properties](#properties) below.

## Properties

<ResponseField name="name" type="string" required>
Name of your project. Used for the global title.

Example: `mintlify`

</ResponseField>

<ResponseField name="navigation" type="Navigation[]" required>
  An array of groups with all the pages within that group
  <Expandable title="Navigation">
    <ResponseField name="group" type="string">
    The name of the group.

    Example: `Settings`

    </ResponseField>
    <ResponseField name="pages" type="string[]">
    The relative paths to the markdown files that will serve as pages.

    Example: `["customization", "page"]`

    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="logo" type="string or object">
  Path to logo image or object with path to "light" and "dark" mode logo images
  <Expandable title="Logo">
    <ResponseField name="light" type="string">
      Path to the logo in light mode
    </ResponseField>
    <ResponseField name="dark" type="string">
      Path to the logo in dark mode
    </ResponseField>
    <ResponseField name="href" type="string" default="/">
      Where clicking on the logo links you to
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="favicon" type="string">
  Path to the favicon image
</ResponseField>

<ResponseField name="colors" type="Colors">
  Hex color codes for your global theme
  <Expandable title="Colors">
    <ResponseField name="primary" type="string" required>
      The primary color. Used for most often for highlighted content, section
      headers, accents, in light mode
    </ResponseField>
    <ResponseField name="light" type="string">
      The primary color used in dark mode.
    </ResponseField>
    <ResponseField name="dark" type="string">
      The primary color used in light mode.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="navbar" type="object">
  Links and the primary button in the top bar. `topbarLinks` and `topbarCtaButton` are not `docs.json` keys.
  <Expandable title="navbar">
    <ResponseField name="links" type="array">
      Each item has `label` and `href`.
    </ResponseField>
    <ResponseField name="primary" type="object">
      The primary button. `type` is `button`, `github`, or `discord`. A button requires `label` and `href`.
    </ResponseField>
  </Expandable>

  ```json
  "navbar": {
    "links": [{ "label": "Support", "href": "mailto:hi@example.com" }],
    "primary": { "type": "button", "label": "Dashboard", "href": "https://example.com" }
  }
  ```
</ResponseField>

<ResponseField name="navigation.versions" type="array">
  Version dropdowns live under `navigation`, not at the top of `docs.json`.
</ResponseField>

<ResponseField name="navigation.global.anchors" type="array">
  External links shown on every page. A top-level `anchors` key is rejected. Each item uses `anchor`, `href`, and `icon`.

  ```json
  "navigation": {
    "global": {
      "anchors": [
        { "anchor": "Documentation", "href": "https://example.com/docs", "icon": "book-open-cover" }
      ]
    }
  }
  ```
</ResponseField>

<ResponseField name="navigation.global" type="object">
  `topAnchor` is not a `docs.json` key. Put the top external link in `navigation.global.anchors`, shown above.
</ResponseField>

<ResponseField name="navigation.tabs" type="array">
  Tabs live under `navigation`, not at the top of `docs.json`. Each item has `tab` and `groups`. See [Navigation](/essentials/navigation).
</ResponseField>

<ResponseField name="api" type="API">
  Configuration for API settings. Learn more about API pages at [API Components](/api-playground/demo).
  <Expandable title="API">
    <ResponseField name="baseUrl" type="string">
      The base url for all API endpoints. If `baseUrl` is an array, it will enable for multiple base url
      options that the user can toggle.
    </ResponseField>

    <ResponseField name="auth" type="Auth">
      <Expandable title="Auth">
        <ResponseField name="method" type='"bearer" | "basic" | "key"'>
          The authentication strategy used for all API endpoints.
        </ResponseField>
        <ResponseField name="name" type="string">
        The name of the authentication parameter used in the API playground.

        If method is `basic`, the format should be `[usernameName]:[passwordName]`
        </ResponseField>
        <ResponseField name="inputPrefix" type="string">
        The default value that's designed to be a prefix for the authentication input field.

        E.g. If an `inputPrefix` of `AuthKey` would inherit the default input result of the authentication field as `AuthKey`.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="playground" type="Playground">
      Configurations for the API playground

      <Expandable title="Playground">
        <ResponseField name="display" default="interactive" type='"interactive" | "simple" | "none" | "auth"'>
          `interactive` is the full playground. `simple` shows the endpoint without extra controls. `none` hides it. `auth` shows it only to signed-in users.

          Learn more at the [playground guides](/api-playground/demo)
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="maintainOrder" type="boolean">
      Enabling this flag ensures that key ordering in OpenAPI pages matches the key ordering defined in the OpenAPI file.

      <Warning>This behavior will soon be enabled by default, at which point this field will be deprecated.</Warning>
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="api.openapi" type="string | string[] | object">
  OpenAPI files live under `api.openapi`, not a top-level `openapi` key. The value is a URL, a relative path, an array of those, or an object that also names overlay documents.

  ```json
  "api": { "openapi": "https://example.com/openapi.json" }
  ```
</ResponseField>

<ResponseField name="footer.socials" type="object">
  Social links live under `footer.socials`. A top-level `footerSocials` key is rejected. Each key is a platform name and each value is a URL.

  ```json
  "footer": {
    "socials": {
      "x": "https://x.com/example",
      "github": "https://github.com/example"
    }
  }
  ```
</ResponseField>

<ResponseField name="feedback" type="Feedback">
  Configurations to enable feedback buttons

  <Expandable title="Feedback">
    <ResponseField name="suggestEdit" type="boolean" default="false">
    Enables a button to allow users to suggest edits via pull requests
    </ResponseField>
    <ResponseField name="raiseIssue" type="boolean" default="false">
    Enables a button to allow users to raise an issue about the documentation
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="appearance" type="object">
  Light and dark mode. `modeToggle` is not a `docs.json` key.
  <Expandable title="appearance">
    <ResponseField name="default" type='"system" | "light" | "dark"' default="system">
      The mode for a new visitor. `system` follows the operating system.
    </ResponseField>
    <ResponseField name="strict" type="boolean" default="false">
      Hide the light/dark toggle. Set `default` to `light` or `dark` when `strict` is true to keep one mode.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="background" type="object">
  Page background. `backgroundImage` is not a `docs.json` key. `image` is the decoration image. `color.light` and `color.dark` are the background colors.
</ResponseField>
